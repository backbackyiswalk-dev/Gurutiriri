-- Store properties only when a feature first changes them; restore on toggle-off/unload.
local function force(group, object, property, value)
local bucket=propertyGroups[group]
if not bucket then bucket={}; propertyGroups[group]=bucket end
local saved=bucket[object]
if not saved then saved={}; bucket[object]=saved end
if saved[property]nil then saved[property]={value=object[property]} end
if object[property]~=value then object[property]=value end
end
local function restore(group)
local bucket=propertyGroups[group]
if not bucket then return end
for object, props in bucket do
for property, saved in props do pcall(function() object[property]=saved.value end) end
end
propertyGroups[group]=nil
end

local function interact(p)
local _,root=body()
if not state.Running or not root or not p or not p.Parent or not p.Enabled then return false end
local pos=promptPosition(p)
if not pos or (root.Position-pos).Magnitude>p.MaxActivationDistance then return false end
if os.clock()-(lastPrompt[p] or -math.huge)<1.5 then return false end
if type(fireproximityprompt)~='function' then return false end
lastPrompt[p]=os.clock()
local ok=pcall(fireproximityprompt,p)
return ok -- Request sent; server success is checked separately for teleports.
end

local function physics(dt)
updateCharacter()
local c,root,h=body()
if not root or not h or h.Health<=0 then stopFlight(); restore('collision'); restore('speed'); return end
local hidden=player:GetAttribute('HidingIn')~=nil or player:GetAttribute('Downed')true
if state.Speed and not hidden then force('speed',h,'WalkSpeed',state.WalkSpeed) else restore('speed') end
if (state.Noclip or state.Fly or os.clock()<state.TeleportUntil) and not hidden then
for _,p in parts do if p.Parent then force('collision',p,'CanCollide',false) end end
else restore('collision') end
if state.Fly and not hidden then
if not flight or flight.Root~=root then
stopFlight()
local a=make('Attachment',{Name='PelicanFlight'},root)
local v=make('LinearVelocity',{Attachment0=a,RelativeTo=Enum.ActuatorRelativeTo.World,
MaxForce=math.huge,VectorVelocity=Vector3.zero},root)
local o=make('AlignOrientation',{Attachment0=a,Mode=Enum.OrientationAlignmentMode.OneAttachment,
MaxTorque=math.huge,Responsiveness=20},root)
flight={Root=root,Attachment=a,Velocity=v,Orientation=o,Move=Vector3.zero}
end
force('flight',h,'AutoRotate',false)
local camera=workspace.CurrentCamera; local move=Vector3.zero
if camera and not UIS:GetFocusedTextBox() then
local cf=camera.CFrame
if UIS:IsKeyDown(Enum.KeyCode.W) then move+=cf.LookVector end
if UIS:IsKeyDown(Enum.KeyCode.S) then move-=cf.LookVector end
if UIS:IsKeyDown(Enum.KeyCode.D) then move+=cf.RightVector end
if UIS:IsKeyDown(Enum.KeyCode.A) then move-=cf.RightVector end
if UIS:IsKeyDown(Enum.KeyCode.Space) then move+=Vector3.yAxis end
if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then move-=Vector3.yAxis end
local look=Vector3.new(cf.LookVector.X,0,cf.LookVector.Z)
if look.Magnitude>.01 then flight.Orientation.CFrame=CFrame.lookAt(Vector3.zero,look) end
end
if move.Magnitude>1 then move=move.Unit end
flight.Move=flight.Move:Lerp(movestate.FlySpeed,1-math.exp(-12dt))
flight.Velocity.VectorVelocity=flight.Move
else stopFlight() end
if state.Repel and not hidden and os.clock()>=state.TeleportUntil then
local target,d=nearest('Monsters')
if target and dstate.MetresPerStud<state.RepelRadius then
local delta=root.Position-state.Entries[target].Anchor.Position
delta=Vector3.new(delta.X,0,delta.Z)
if delta.Magnitude>.01 then
local velocity=delta.Unitstate.RepelSpeed+Vector3.new(0,math.max(root.AssemblyLinearVelocity.Y,8),0)
if flight then flight.Velocity.VectorVelocity=velocity else root.AssemblyLinearVelocity=velocity end
end
end
end
end

-- Store properties only when a feature first changes them; restore on toggle-off/unload.
local function force(group, object, property, value)
local bucket=propertyGroups[group]
if not bucket then bucket={}; propertyGroups[group]=bucket end
local saved=bucket[object]
if not saved then saved={}; bucket[object]=saved end
if saved[property]nil then saved[property]={value=object[property]} end
if object[property]~=value then object[property]=value end
end
local function restore(group)
local bucket=propertyGroups[group]
if not bucket then return end
for object, props in bucket do
for property, saved in props do pcall(function() object[property]=saved.value end) end
end
propertyGroups[group]=nil
end
