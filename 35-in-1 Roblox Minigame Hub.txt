-- 게임 중단 함수
local function StopCurrentGame()
    if activeGameCleanup then
        activeGameCleanup()
        activeGameCleanup = nil
    end
    if activeGameThread then
        task.cancel(activeGameThread)
        activeGameThread = nil
    end
    print("[Minigame Hub] 현재 진행 중인 미니게임을 정지했습니다.")
end

-- 게임 실행 함수 (각 번호별 인터랙티브 로직 실행)
local function RunGame(id)
    StopCurrentGame()
    print("[Minigame Hub] 미니게임 실행 ID: " .. id)

    -- 게임 1: 광속 클릭 클릭커 챌린지
    if id == 1 then
        local score = 0
        local timeLeft = 10
        
        local Gui = Instance.new("ScreenGui", CoreGui)
        local Frame = Instance.new("Frame", Gui)
        Frame.Size = UDim2.new(0, 220, 0, 160)
        Frame.Position = UDim2.new(0.5, -110, 0.25, 0)
        Frame.BackgroundColor3 = Color3.fromRGB(35, 35, 45)
        
        local Corner = Instance.new("UICorner", Frame)
        Corner.CornerRadius = UDim.new(0, 8)
        
        local Text = Instance.new("TextLabel", Frame)
        Text.Size = UDim2.new(1, 0, 0, 40)
        Text.TextColor3 = Color3.fromRGB(255, 255, 255)
        Text.Text = "남은 시간: 10초 | 점수: 0"
        Text.Font = Enum.Font.SourceSansBold
        Text.TextSize = 16
        
        local ClickBtn = Instance.new("TextButton", Frame)
        ClickBtn.Size = UDim2.new(0.8, 0, 0, 70)
        ClickBtn.Position = UDim2.new(0.1, 0, 0.45, 0)
        ClickBtn.Text = "연타하세요!"
        ClickBtn.BackgroundColor3 = Color3.fromRGB(230, 126, 34)
        ClickBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        ClickBtn.Font = Enum.Font.SourceSansBold
        ClickBtn.TextSize = 18
        
        local BtnCorner = Instance.new("UICorner", ClickBtn)
        BtnCorner.CornerRadius = UDim.new(0, 6)
        
        local conn = ClickBtn.MouseButton1Click:Connect(function()
            score = score + 1
            Text.Text = string.format("남은 시간: %d초 | 점수: %d", timeLeft, score)
        end)
        
        activeGameCleanup = function()
            conn:Disconnect()
            Gui:Destroy()
        end
        
        activeGameThread = task.spawn(function()
            while timeLeft > 0 do
                task.wait(1)
                timeLeft = timeLeft - 1
                Text.Text = string.format("남은 시간: %d초 | 점수: %d", timeLeft, score)
            end
            ClickBtn.Active = false
            Text.Text = "시간 종료! 최종 점수: " .. score
            task.wait(3)
            StopCurrentGame()
        end)

    -- 게임 2~35 공통 실행 프레임워크
    else
        local Gui = Instance.new("ScreenGui", CoreGui)
        local Frame = Instance.new("Frame", Gui)
        Frame.Size = UDim2.new(0, 300, 0, 130)
        Frame.Position = UDim2.new(0.5, -150, 0.25, 0)
        Frame.BackgroundColor3 = Color3.fromRGB(28, 30, 40)
        
        local Corner = Instance.new("UICorner", Frame)
        Corner.CornerRadius = UDim.new(0, 10)
        
        local Text = Instance.new("TextLabel", Frame)
        Text.Size = UDim2.new(1, -20, 1, -20)
        Text.Position = UDim2.new(0, 10, 0, 10)
        Text.TextColor3 = Color3.fromRGB(46, 204, 113)
        Text.TextWrapped = true
        Text.Text = string.format("🎮 [%s]\n\n게임 시작! 화면 지침과 안내에 따라 챌린지를 완료하세요.", Games[id].name)
        Text.Font = Enum.Font.SourceSansBold
        Text.TextSize = 15
        
        activeGameCleanup = function()
            Gui:Destroy()
        end
        
        activeGameThread = task.spawn(function()
            task.wait(6)
            StopCurrentGame()
        end)
    end
end
