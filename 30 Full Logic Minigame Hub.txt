local function StopCurrentGame()
    if activeGameCleanup then
        pcall(activeGameCleanup)
        activeGameCleanup = nil
    end
    if activeGameThread then
        task.cancel(activeGameThread)
        activeGameThread = nil
    end
end

-- 1. 스피드 연타 클릭커
GameLogics[1] = function()
    local Gui, Frame, Content = CreateCanvas("1. 스피드 연타 클릭커")
    local clicks = 0
    local timeLeft = 10
    local Label = Instance.new("TextLabel", Content)
    Label.Size = UDim2.new(1, 0, 0, 30)
    Label.Text = "남은 시간: 10초 | 클릭: 0"
    Label.TextColor3 = Color3.fromRGB(255, 255, 255)
    Label.TextSize = 16
    local Btn = Instance.new("TextButton", Content)
    Btn.Size = UDim2.new(0.8, 0, 0, 80)
    Btn.Position = UDim2.new(0.1, 0, 0.35, 0)
    Btn.BackgroundColor3 = Color3.fromRGB(52, 152, 219)
    Btn.Text = "연타!!"
    Btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    Btn.TextSize = 22
    Instance.new("UICorner", Btn).CornerRadius = UDim.new(0, 8)
    
    local conn = Btn.MouseButton1Click:Connect(function()
        clicks = clicks + 1
        Label.Text = string.format("남은 시간: %d초 | 클릭: %d", timeLeft, clicks)
    end)
    activeGameCleanup = function() conn:Disconnect() Gui:Destroy() end
    activeGameThread = task.spawn(function()
        while timeLeft > 0 do
            task.wait(1)
            timeLeft = timeLeft - 1
            Label.Text = string.format("남은 시간: %d초 | 클릭: %d", timeLeft, clicks)
        end
        Btn.Active = false
        Label.Text = "종료! 최종 기록: " .. clicks .. "회"
    end)
end

            state = "finished"
            Btn.BackgroundColor3 = Color3.fromRGB(149, 165, 166)
            Btn.Text = "너무 빨랐습니다! 실패"

            Btn.Text = "반응속도: " .. ms .. " ms !"

    end)
    activeGameCleanup = function() conn:Disconnect() Gui:Destroy() end
